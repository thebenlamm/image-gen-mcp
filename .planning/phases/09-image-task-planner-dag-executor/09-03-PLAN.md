    **C. Add `handleImageTask` at the bottom of the file after `handleImageOp` (around current line 794):**
    ```typescript
    export interface ImageTaskArgs {
      goal: string;
      input_images?: string[];
      constraints?: TaskConstraints;
      dry_run?: boolean;
      runId?: string;
      seed?: number;
      outputDir?: string;
      outputPath?: string;
    }

    export async function handleImageTask(args: ImageTaskArgs): Promise<{
      content: Array<{ type: 'text'; text: string }>;
    }> {
      const startedAt = Date.now();
      const runId = args.runId ?? createRunId();
      let runDir: string;
      try {
        runDir = await resolveRunDir(runId);
      } catch (err) {
        return imageTaskErrorResponse(runId, `Failed to create run directory: ${err instanceof Error ? err.message : String(err)}`);
      }

      // Build registry snapshot for planner (mirror handleListCapabilities serializer).
      const registrySnapshot = capabilityRegistry.list().map((capability) => ({
        op: capability.op,
        provider: capability.provider,
        modelVersion: capability.modelVersion,
        constraints: capability.constraints,
        cost: capability.cost,
        latencyMsP50: capability.latencyMsP50,
        quality: capability.quality,
      }));

      // 1. Plan
      let plan: Plan;
      try {
        plan = await planImageTask({
          goal: args.goal,
          inputImages: args.input_images,
          constraints: args.constraints,
          registrySnapshot,
          seed: args.seed,
        });
      } catch (err) {
        return imageTaskPlannerErrorResponse(runId, err);
      }

      // 2. Validate (BUDGET_CAP_EXCEEDED, capability params, IMAGE_GEN_INPUT_ROOT, etc.)
      try {
        await validatePlan(plan, { registry: capabilityRegistry, constraints: args.constraints });
      } catch (err) {
        return imageTaskPlannerErrorResponse(runId, err);  // PlannerError carries code:'BUDGET_CAP_EXCEEDED', estimated_cost_usd, cap_usd
      }

      // 3. Dry-run early return — no provider invocation
      if (args.dry_run === true) {
        const drySerialized = serializeImageTaskResponse({
          plan,
          dagResult: {
            status: 'success',
            nodes: [],
            skips: [],
            totals: { costUsd: plan.estimatedCostUsd, latencyMs: plan.estimatedLatencyMs },
          },
          runId,
        });
        return { content: [{ type: 'text', text: JSON.stringify({ ...drySerialized, dry_run: true }) }] };
      }

      // 4. Build inputs map
      const inputImages: Record<string, string> = {};
      (args.input_images ?? []).forEach((p, i) => { inputImages[`image_${i}`] = p; });

      // 5. Execute DAG
      let dagResult;
      try {
        dagResult = await executeDag({ plan, inputImages, runId, runDir });
      } catch (err) {
        return imageTaskErrorResponse(runId, err instanceof Error ? err.message : String(err));
      }

      // 6. Save terminal image via standard output rules (only when terminal succeeded)
      if (dagResult.terminalOutput) {
        const finalPath = await resolveOutputPath({
          outputPath: args.outputPath,
          outputDir: args.outputDir,
          prompt: args.goal,
          provider: 'image_task',
        });
        // Read the terminal artifact buffer once and saveImage to user-facing path.
        const fs = await import('node:fs/promises');
        const buf = await fs.readFile(dagResult.terminalOutput.path);
        await saveImage(buf, finalPath);
        dagResult.terminalOutput.path = finalPath;
      }

        });

        // Manifest write failure is non-fatal for the response; surface in trace metadata.
        console.error(`[image_task] manifest write failed: ${err instanceof Error ? err.message : String(err)}`);
      }

      // 8. Serialize compact path-only response (D-17, D-18, D-20)
      try {
        const response = serializeImageTaskResponse({ plan, dagResult, runId });

          // Defensive: this should never trigger if executor is correct.
          return imageTaskErrorResponse(runId, `Response guard tripped: ${err.message}`);
        }
        throw err;
      }
    }

    function imageTaskErrorResponse(runId: string, message: string): { content: Array<{ type: 'text'; text: string }> } {
      return {
        content: [{
          type: 'text',
          text: JSON.stringify({ success: false, error: { message }, runId, trace: [] }),
        }],
      };
    }

    function imageTaskPlannerErrorResponse(runId: string, err: unknown): { content: Array<{ type: 'text'; text: string }> } {
      if (err instanceof PlannerError) {
        return {
          content: [{
            type: 'text',
    ```
