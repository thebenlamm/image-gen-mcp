<!-- Adapters must match this shape exactly. -->

```typescript
export interface Capability {
  op: CapabilityOp;                  // 'generate' for these files
  provider: string;                  // 'openai' | 'gemini' | 'together'
  modelVersion: string;
  constraints: CapabilityConstraints;
  cost: CapabilityCost;              // { perCallUsd: number }
  latencyMsP50?: number;
  quality?: CapabilityQuality;       // MUST include unscoredJustification (non-empty string)
  invoke: CapabilityInvoke;
}

export interface CapabilityConstraints {
  maxPromptLength?: number;
  supportedSizes?: Array<'square' | 'landscape' | 'portrait'>;
  requiresInputImage?: boolean;
  supportsMultipleInputs?: boolean;
  outputFormat?: 'png';
}

export interface CapabilityQuality {
  scores?: Record<string, number>;
  evaluatedAt?: string;
  evalResultPath?: string;
  unscoredJustification?: string;    // REQUIRED when registered with allowUnscoredProduction=true
}

// invoke() must return:
// { kind: 'image', buffer: Buffer, model: string, revisedPrompt?: string, metadata?: Record<string, unknown> }

export class CapabilityInvokeError extends Error {
  constructor(
    public readonly code: 'INPUT_TOO_LARGE' | 'CONSTRAINT_VIOLATION' | 'PROVIDER_FAILURE' | 'TIMEOUT' | 'UNSUPPORTED',
    message: string,
    public readonly retryable: boolean,
    public readonly suggestion?: string,
  );
}
```

export interface ImageProvider {
  name: ProviderName;
  defaultModel: string;
  supportsSize: boolean;
  generate(params: GenerateParams): Promise<GenerateResult>;

export interface GenerateParams {

  model?: string;
  size?: 'square' | 'landscape' | 'portrait';
}

export interface GenerateResult {
  buffer: Buffer;
  model: string;
  revisedPrompt?: string;
}
