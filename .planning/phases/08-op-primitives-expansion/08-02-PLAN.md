    ```ts
    import { createWorker } from 'tesseract.js';

    type Worker = Awaited<ReturnType<typeof createWorker>>;

    const pool = new Map<string, Promise<Worker>>();

    /**
     * Per-call mode: create → recognize → terminate. Used by eval scoring
     * (scoreOcrTextPresence) where tesseract worker reuse is not a hot path.
     */
    export async function recognizeOnce(path: string, lang = 'eng') {
      const worker = await createWorker(lang);
      try {
        const result = await worker.recognize(path);
        return result.data;
      } finally {
        await worker.terminate().catch(() => undefined);
      }
    }

    /**
     * Pooled mode: cache one worker per language indefinitely (server lifetime).
     * Used by analyze_ocr capability where ~20 invocations save ~30s of init.
     * Pool eviction is OUT OF SCOPE for Phase 8 (D-32 deferred ideas).
     */
    export async function recognizePooled(path: string, lang = 'eng') {
      let workerPromise = pool.get(lang);
      if (!workerPromise) {
        workerPromise = createWorker(lang);
        pool.set(lang, workerPromise);
      }
      const worker = await workerPromise;
      const result = await worker.recognize(path);
      return result.data;
    }

    /**
     * Optional shutdown hook. NOT wired in Phase 8 (the MCP server has no
     * SIGTERM hook today); exposed for tests and a future Phase 9 wiring.
     */
    export async function terminatePool(): Promise<void> {
      const promises = Array.from(pool.values());
      pool.clear();
      for (const promise of promises) {
        const w = await promise.catch(() => null);
        if (w) await w.terminate().catch(() => undefined);
      }
    }
    ```
